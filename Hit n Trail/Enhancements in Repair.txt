import json
import shlex
import subprocess


class RepairPlanner:
    SUPPORTED_WORKLOADS = {
        "Deployment",
        "StatefulSet",
        "DaemonSet",
    }

    def __init__(self):
        self._owner_cache = {}
        self._planned = set()

    def _kubectl_json(self, *args):
        result = subprocess.run(
            ["kubectl", *args, "-o", "json"],
            capture_output=True,
            text=True,
            check=False,
        )

        if result.returncode != 0:
            return None

        try:
            return json.loads(result.stdout)
        except json.JSONDecodeError:
            return None

    def _resolve_workload(self, resource):
        namespace = resource.get("namespace")
        name = resource.get("resource") or resource.get("name")
        kind = resource.get("kind", "")

        if not namespace or not name:
            return None

        # A known controller can be used directly.
        if kind.lower() in {
            "deployment", "statefulset", "daemonset"
        }:
            actual_kind = kind.title()
            return {
                "kind": actual_kind,
                "name": name,
                "namespace": namespace,
            }

        cache_key = (namespace, name)
        if cache_key in self._owner_cache:
            return self._owner_cache[cache_key]

        # Get the actual Pod from Kubernetes.
        pod = self._kubectl_json(
            "get", "pod", name, "-n", namespace
        )

        if not pod:
            self._owner_cache[cache_key] = None
            return None

        owner_refs = pod.get("metadata", {}).get(
            "ownerReferences", []
        )
        if not owner_refs:
            self._owner_cache[cache_key] = None
            return None

        owner = owner_refs[0]
        owner_kind = owner.get("kind")
        owner_name = owner.get("name")

        # A ReplicaSet is normally owned by a Deployment.
        if owner_kind == "ReplicaSet":
            replica_set = self._kubectl_json(
                "get", "replicaset", owner_name,
                "-n", namespace,
            )

            if not replica_set:
                self._owner_cache[cache_key] = None
                return None

            rs_owners = replica_set.get(
                "metadata", {}
            ).get("ownerReferences", [])

            if not rs_owners:
                self._owner_cache[cache_key] = None
                return None

            owner = rs_owners[0]
            owner_kind = owner.get("kind")
            owner_name = owner.get("name")

        if owner_kind not in self.SUPPORTED_WORKLOADS:
            self._owner_cache[cache_key] = None
            return None

        resolved = {
            "kind": owner_kind,
            "name": owner_name,
            "namespace": namespace,
        }

        self._owner_cache[cache_key] = resolved
        return resolved

    @staticmethod
    def _target_args(target):
        return (
            f"{target['kind'].lower()}/{target['name']}"
        )

    def _blocked(self, category, resource, reason):
        return {
            "action": f"{category} repair blocked",
            "resource": {
                "kind": resource.get("kind", "Pod"),
                "name": (
                    resource.get("resource")
                    or resource.get("name")
                ),
                "namespace": resource.get("namespace"),
            },
            "reason": reason,
            "command": None,
            "rollback": None,
            "risk": "UNKNOWN",
            "status": "BLOCKED",
        }

    def create_plan(self, root_causes):
        actions = []
        self._owner_cache.clear()
        self._planned.clear()

from rootiq.engine.rule_context import RuleContext
from rootiq.rules.base import BaseRule


class DeploymentImageTagRule(BaseRule):

    id = "DEPLOYMENT-005"

    name = "Deployment Image Tag Validation"

    description = (
        "Detect deployments using mutable or missing container image tags."
    )

    resource_type = "deployment"

    severity = "medium"

    category = "deployment"

    def evaluate(self, context: RuleContext):

        mutable_tags = {
            "latest",
            "dev",
            "development",
            "snapshot",
            "test",
            "master",
            "main",
            "edge",
            "nightly",
        }

        for deployment in context.resources:

            deployment_name = deployment["name"]
            namespace = deployment["namespace"]

            for container in deployment.get("containers", []):

                image = container.get("image", "")

                #
                # Image has no explicit tag
                #
                if ":" not in image:

                    context.report(
                        rule_id=self.id,
                        severity="high",
                        title="Container image has no explicit tag",
                        resource=deployment_name,
                        namespace=namespace,
                        description=(
                            f"Container '{container['name']}' uses image "
                            f"'{image}' without an explicit tag."
                        ),
                        recommendation=(
                            "Specify an immutable image tag "
                            "(example: nginx:1.27.2)."
                        ),
                        metadata={
                            "container": container["name"],
                            "image": image,
                        },
                    )

                    continue

                tag = image.rsplit(":", 1)[1].lower()

                #
                # Mutable image tag
                #
                if tag in mutable_tags:

                    context.report(
                        rule_id=self.id,
                        severity="medium",
                        title="Container uses mutable image tag",
                        resource=deployment_name,
                        namespace=namespace,
                        description=(
                            f"Container '{container['name']}' uses the "
                            f"mutable image tag '{tag}'."
                        ),
                        recommendation=(
                            "Use an immutable version tag "
                            "(example: v1.4.2 or 1.27.2) "
                            "to ensure reproducible deployments."
                        ),
                        metadata={
                            "container": container["name"],
                            "image": image,
                            "tag": tag,
                        },

    def create_plan(self, root_causes):
        actions = []
        self._owner_cache.clear()
        self._planned.clear()

        for cause in root_causes:
            category = str(cause.get("category", "")).upper()

            if category not in {"IMAGE", "RESOURCE", "POD_HEALTH"}:
                continue

            for resource in cause.get("resources", []):
                target = self._resolve_workload(resource)

                if not target:
                    actions.append(self._blocked(
                        category,
                        resource,
                        "Could not resolve the Pod's owning "
                        "Deployment, StatefulSet, or DaemonSet "
                        "from Kubernetes. No command was generated.",
                    ))

                workload = self._target_args(target)
                container = resource.get("container")

                if category in {"IMAGE", "RESOURCE"} and not container:
                    actions.append(self._blocked(
                        category,
                        resource,
                        "Container name is missing; cannot safely "
                        "target the affected container.",

                dedupe_key = (

class RootCauseEngine:

    name = "RootCauseEngine"


    def analyze(self, issues):

        grouped = defaultdict(list)


        for issue in issues:

            category = self.detect_category(issue)

            grouped[category].append(issue)



        root_causes = []


        for category, items in grouped.items():

            if category == "IMAGE":

                root_causes.append(
                    self.build_image_rootcause(items)
                )


            elif category == "RESOURCE":

                root_causes.append(
                    self.build_resource_rootcause(items)
                )


            elif category == "POD_HEALTH":

                root_causes.append(
                    self.build_restart_rootcause(items)
                )


        return root_causes

    def _kubectl_json(self, *args):
        result = subprocess.run(
            ["kubectl", *args, "-o", "json"],
            capture_output=True,
            text=True,
            check=False,
        )

        if result.returncode != 0:
            return None

        try:
            return json.loads(result.stdout)
        except json.JSONDecodeError:
            return None

    def _resolve_workload(self, resource):
        namespace = resource.get("namespace")
        name = resource.get("resource") or resource.get("name")
        kind = resource.get("kind", "")

        if not namespace or not name:
            return None

        # A known controller can be used directly.
        if kind.lower() in {
            "deployment", "statefulset", "daemonset"
        }:
            actual_kind = kind.title()
            return {
                "kind": actual_kind,
                "name": name,
                "namespace": namespace,
            }

        cache_key = (namespace, name)
        if cache_key in self._owner_cache:
            return self._owner_cache[cache_key]

        # Get the actual Pod from Kubernetes.
        pod = self._kubectl_json(
            "get", "pod", name, "-n", namespace
        )

        if not pod:
            self._owner_cache[cache_key] = None
            return None

        owner_refs = pod.get("metadata", {}).get(
            "ownerReferences", []
        )
        if not owner_refs:
            self._owner_cache[cache_key] = None
            return None

        owner = owner_refs[0]
        owner_kind = owner.get("kind")
        owner_name = owner.get("name")

        # A ReplicaSet is normally owned by a Deployment.
        if owner_kind == "ReplicaSet":
            replica_set = self._kubectl_json(
                "get", "replicaset", owner_name,
                "-n", namespace,
            )

            if not replica_set:

                return None

                "metadata", {}
            ).get("ownerReferences", [])

            if not rs_owners:
                self._owner_cache[cache_key] = None
                return None

            owner = rs_owners[0]
            owner_kind = owner.get("kind")

from collections import defaultdict


class RootCauseEngine:

    name = "RootCauseEngine"


    def analyze(self, issues):

        grouped = defaultdict(list)


        for issue in issues:

            category = self.detect_category(issue)

            grouped[category].append(issue)



        root_causes = []


        for category, items in grouped.items():

            if category == "IMAGE":

                root_causes.append(
                    self.build_image_rootcause(items)
                )


            elif category == "RESOURCE":

                root_causes.append(
                    self.build_resource_rootcause(items)
                )


            elif category == "POD_HEALTH":

                root_causes.append(
                    self.build_restart_rootcause(items)
                )


        return root_causes




    def detect_category(self, issue):
        issue_id = issue.get("id", "")
        title = issue.get("title", "").lower()
        description = issue.get("description", "").lower()
        labels = issue.get("labels") or {}

        # Image-pull failures must be handled as image problems,
        # not generic pod-health/restart problems.
        if (
            issue_id == "POD-003"
            or labels.get("reason") == "ImagePullBackOff"
            or "imagepullbackoff" in description
            or "errimagepull" in description
        ):
            return "IMAGE"

        if (
            "image" in title
            or issue_id in {"DEPLOYMENT-005", "POD-009"}
        ):
            return "IMAGE"

        if (
            "resource" in title
            or "cpu" in title
            or "memory" in title
            or issue_id == "DEPLOYMENT-006"
        ):
            return "RESOURCE"

        if (
            "restart" in title
            or "crash" in title
        ):
            return "POD_HEALTH"

        return "UNKNOWN"

from rootiq.config.config_loader import RootIQConfig
import json
import shlex
import subprocess


class RepairPlanner:
    SUPPORTED_WORKLOADS = {
        "Deployment",
        "StatefulSet",
        "DaemonSet",
    }

    def __init__(self):
        self._owner_cache = {}
        self._planned = set()
        self.config = RootIQConfig()

    def _kubectl_json(self, *args):
        result = subprocess.run(
            ["kubectl", *args, "-o", "json"],
            capture_output=True,
            text=True,
            check=False,
        )

        if result.returncode != 0:
            return None

        try:
            return json.loads(result.stdout)
        except json.JSONDecodeError:
            return None

    def _resolve_workload(self, resource):
        namespace = resource.get("namespace")
        name = resource.get("resource") or resource.get("name")
        kind = resource.get("kind", "")

        if not namespace or not name:
            return None

        # A known controller can be used directly.
        if kind.lower() in {
            "deployment", "statefulset", "daemonset"
        }:
            actual_kind = kind.title()
            return {
                "kind": actual_kind,
                "name": name,
                "namespace": namespace,
            }

        cache_key = (namespace, name)
        if cache_key in self._owner_cache:
            return self._owner_cache[cache_key]

        # Get the actual Pod from Kubernetes.
        pod = self._kubectl_json(
            "get", "pod", name, "-n", namespace
        )

        if not pod:
            self._owner_cache[cache_key] = None
            return None

        owner_refs = pod.get("metadata", {}).get(
            "ownerReferences", []
        )
        if not owner_refs:
            self._owner_cache[cache_key] = None
            return None

        owner = owner_refs[0]
        owner_kind = owner.get("kind")
        owner_name = owner.get("name")

        # A ReplicaSet is normally owned by a Deployment.
        if owner_kind == "ReplicaSet":
            replica_set = self._kubectl_json(
                "get", "replicaset", owner_name,
                "-n", namespace,
            )

            if not replica_set:
                self._owner_cache[cache_key] = None
                return None

            rs_owners = replica_set.get(
                "metadata", {}
            ).get("ownerReferences", [])

            if not rs_owners:
                self._owner_cache[cache_key] = None
                return None

            owner = rs_owners[0]
            owner_kind = owner.get("kind")
            owner_name = owner.get("name")

        if owner_kind not in self.SUPPORTED_WORKLOADS:
            self._owner_cache[cache_key] = None
            return None

        resolved = {
            "kind": owner_kind,
            "name": owner_name,
            "namespace": namespace,
        }

class IncidentLoader:

    def load_latest(self):

        latest_file = (
            self.incident_dir /
            "latest.json"
        )


        if not latest_file.exists():

            raise FileNotFoundError(
                "latest.json not found"
            )

            latest_file,
